JS

const btn = document.getElementById('showSwatches');
const popup = document.getElementById('swatchPopup');
const closeBtn = document.getElementById('closePopup');
const select = document.getElementById('colorSelect');
const container = document.getElementById('swatchContainer');

// Open popup and generate swatches
btn.addEventListener('click', () => {
  container.innerHTML = ''; // Clear previous swatches
  
  // Loop through all options in the select element
  Array.from(select.options).forEach(option => {
    const colorValue = option.value;
    
    // Create wrapper
    const item = document.createElement('div');
    item.className = 'swatch-item';
    
    // Create color box
    const box = document.createElement('span');
    box.className = 'swatch-box';
    box.style.backgroundColor = colorValue;
    
    // Create text label
    const label = document.createElement('span');
    label.textContent = option.text;
    
    // Append to container
    item.appendChild(box);
    item.appendChild(label);
    container.appendChild(item);
  });
  
  popup.style.display = 'flex';
});

// Close popup on 'X' click
closeBtn.addEventListener('click', () => {
  popup.style.display = 'none';
});

// Close popup when clicking outside content
window.addEventListener('click', (e) => {
  if (e.target === popup) {
    popup.style.display = 'none';
  }
});
